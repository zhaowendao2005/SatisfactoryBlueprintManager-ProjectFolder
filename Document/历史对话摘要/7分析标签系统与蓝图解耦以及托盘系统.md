### 1. 全局标签Store核心逻辑

```typescript
// stores/GlobalTags/index.ts
export const useGlobalTagsStore = defineStore('globalTags', {
  state: (): GlobalTagsState => ({
    tags: [],
    blueprintTagsMap: new Map<string, string[]>()
  }),
  
  actions: {
    async initializeGlobalTags() {
      const config = await this.datasource.loadGlobalTags()
      this.tags = config.tags || []
      this.blueprintTagsMap = new Map(
        Object.entries(config.blueprintTags || {})
      )
    },
    
    createTag(tag: TagDefinition) {
      this.tags.push(tag)
      this.saveGlobalTags()
    },
    
    batchAddTags(blueprintPath: string, tagIds: string[]) {
      const existing = this.blueprintTagsMap.get(blueprintPath) || []
      const newTags = [...new Set([...existing, ...tagIds])]
      this.blueprintTagsMap.set(blueprintPath, newTags)
      this.saveGlobalTags()
    }
  }
})
```

```typescript
  const globalTagsStore = useGlobalTagsStore()
```

### 3. 多标签逻辑筛选

```typescript
// utils/tagHelpers.ts
export function filterBlueprintsByTags(
  blueprints: ActiveBlueprintNodeWithTags[],
  activeTags: Map<string, 'and' | 'or' | 'not'>
): ActiveBlueprintNodeWithTags[] {
  if (activeTags.size === 0) return blueprints

  return blueprints.filter((blueprint) => {
    // 分组：与、或、非
    const andTags: string[] = []
    const orTags: string[] = []
    const notTags: string[] = []

    activeTags.forEach((logic, tagId) => {
      if (logic === 'and') andTags.push(tagId)
      else if (logic === 'or') orTags.push(tagId)
      else if (logic === 'not') notTags.push(tagId)
    })

    // 1. 非(NOT)逻辑：不能匹配任何"非"标签
    if (notTags.length > 0) {
      if (notTags.some(tagId => matchesTag(blueprint, tagId))) {
        return false
      }
    }

    // 2. 与(AND)逻辑：必须匹配所有"与"标签
    if (andTags.length > 0) {
      if (!andTags.every(tagId => matchesTag(blueprint, tagId))) {
        return false
      }
    }

    // 3. 或(OR)逻辑：需要匹配至少一个"或"标签
    if (orTags.length > 0) {
      if (!orTags.some(tagId => matchesTag(blueprint, tagId))) {
        return false
      }
    }

    return true
  })
}
```

## 踩坑与经验总结

### 坑1：导入路径错误
问题：`GlobalTagsHandler.ts` 中导入路径错误
```typescript
// 错误
import { GlobalTagsFileService } from '../GlobalTags/file-service'

// 正确
import { GlobalTagsFileService } from '../Service/GlobalTags/file-service'
```
原因：文件实际位置在 `Service/GlobalTags/` 而非 `GlobalTags/`
解决：修正导入路径
经验：创建新文件时注意目录结构，导入路径要与实际位置一致

### 坑2：Element Plus组件导入错误
问题：`element-plus/es/components/split/style/css` 导入失败
原因：Element Plus的组件样式导入方式不正确
解决：改用正确的导入方式或使用全局样式
经验：使用第三方UI库时，注意样式导入的正确方式

### 坑3：双重滚动条冲突
问题：外层和内层滚动条冲突，无法正常滚动
原因：CSS布局问题，容器高度计算错误
解决：明确设置容器高度，使用 `overflow-y: auto` 实现独立滚动
经验：多层滚动容器需要明确高度和overflow属性

### 坑4：框选性能问题
问题：框选时卡顿，特别是蓝图数量多时
原因：每次mousemove都计算所有图标位置
解决：
1. 缓存图标位置（开始框选时一次性计算）
2. 使用 `requestAnimationFrame` 节流
3. 松手时统一计算选中状态
经验：高频事件需要节流，批量计算优于即时计算

### 坑5：框选与点击冲突
问题：在图标上拖动时无法触发框选
原因：图标点击事件阻止了框选事件
解决：
1. 记录mousedown位置
2. 监听mousemove，移动距离>5px时标记为拖动
3. 拖动时不触发点击事件
经验：区分点击和拖动需要记录初始位置和移动距离

### 坑6：图标导入错误
问题：`Battery`、`Truck` 等图标不存在
原因：Element Plus Icons中不存在这些图标名称
解决：使用实际存在的图标名称或使用其他图标
经验：使用图标库时，先确认图标名称是否存在

### 坑7：字体大小计算逻辑
问题：字体大小计算不准确，无法正确显示
原因：字符宽度估算不准确，未考虑中英文混合
解决：
1. 使用字符宽度系数0.6（考虑中英文混合）
2. 设置字体大小范围（8-11px）
3. 达到最低仍无法显示时启用溢出处理
经验：字体大小计算需要考虑字符类型和容器限制

### 坑8：上级标签筛选不包含下级
问题：使用上级标签筛选时，无法匹配下级标签的蓝图
原因：筛选逻辑只检查精确匹配
解决：实现递归匹配逻辑，检查路径前缀
经验：树形结构的筛选需要考虑层级关系

## 架构决策记录

### 决策1：采用全局标签系统而非混合方案
原因：
- 用户明确要求全面重构，不考虑向后兼容
- 全局标签系统架构更清晰，职责分离
- 便于后续扩展和维护

影响：
- 需要创建新的配置文件
- 需要修改所有标签相关的代码
- 用户需要重新配置标签

### 决策2：使用委托模式保持接口兼容
原因：
- 减少对现有代码的影响
- 保持API稳定性
- 便于渐进式迁移

影响：
- ActiveBlueprintStore需要维护委托方法
- 代码复杂度略有增加

### 决策3：标签关系基于blueprintPath而非nodeId
原因：
- 实现跨配置复用
- 蓝图路径是唯一标识
- 更符合业务逻辑

影响：
- 需要将nodeId转换为blueprintPath
- 路径变化时需要更新标签关系
