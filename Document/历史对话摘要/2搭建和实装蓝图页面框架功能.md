```typescript
// src-electron/Service/Blueprint/file-scanner.ts
async function scanDirectory(
  dirPath: string,
  maxDepth: number = 5,
  currentDepth: number = 0,
  fileCount: { count: number } = { count: 0 }
): Promise<BlueprintNode> {
  if (currentDepth > maxDepth) {
    return createDirectoryNode(dirPath, [])
  }
  
  if (fileCount.count > 500) {
    throw new Error('文件数超过限制（500个）')
  }
  
  const entries = await fs.readdir(dirPath, { withFileTypes: true })
  const children: BlueprintNode[] = []
  
  for (const entry of entries) {
    const fullPath = path.join(dirPath, entry.name)
    
    if (entry.isDirectory()) {
      const childNode = await scanDirectory(fullPath, maxDepth, currentDepth + 1, fileCount)
      children.push(childNode)
    } else if (entry.name.endsWith('.sbp')) {
      // 仅处理 .sbp 文件，跳过 .sbpcfg
      children.push(createBlueprintNode(fullPath))
      fileCount.count++
    }
  }
  
  return createDirectoryNode(dirPath, children)
}
```

```typescript
async getChildren(nodeId: string): Promise<BlueprintNode[]> {
  // 规范化路径（确保与缓存键一致）
  const normalizedNodeId = nodeId.replace(/\\/g, '/')
  
  // 检查缓存
  if (this.childrenCache.has(normalizedNodeId)) {
    return this.childrenCache.get(normalizedNodeId)!
  }
  
  // 调用 IPC 读取单层目录
  const children = await this.getBlueprintAPI().readDirectoryShallow(normalizedNodeId)
  this.childrenCache.set(normalizedNodeId, children)
  return children
}
```

## ⚠️ 踩的坑与经验总结

### 坑1：运算符优先级问题

**问题**：
```typescript
// 错误代码
const sourceName = existingConfig?.sourceName ?? getBasename(selectedPath) || selectedPath
```

**现象**：编译错误 `Cannot use "||" with "??" without parentheses`

**原因**：`??`（空值合并）和 `||`（逻辑或）的优先级不同，需要括号明确优先级

**解决方案**：
```typescript
// 正确代码
const sourceName = existingConfig?.sourceName ?? (getBasename(selectedPath) || selectedPath)
```

**经验总结**：混用 `??` 和 `||` 时，必须用括号明确优先级

---

### 坑2：el-tree 懒加载与全量扫描冲突

**问题**：添加源后，树结构无法正常显示

**现象**：扫描返回了完整的树结构，但 el-tree 不显示子节点

**原因**：el-tree 的 `lazy` 模式会忽略节点的 `children` 属性，只通过 `load` 函数加载

**解决方案**：
1. 将根节点的 `children` 设为 `undefined`
2. 将扫描到的子节点缓存到 `childrenCache`
3. 在 `loadNode` 函数中从缓存获取子节点

**经验总结**：使用 el-tree 的 `lazy` 模式时，不要设置 `children` 属性，而是通过 `load` 函数提供数据

---

### 坑3：路径规范化不一致导致缓存失效

**问题**：添加源后，展开目录节点无法加载子节点

**现象**：缓存键与节点 ID 路径格式不一致（`\` vs `/`）

**原因**：Windows 路径使用反斜杠，但存储时转换为正斜杠，导致缓存键不匹配

**解决方案**：
```typescript
// 统一规范化路径
const normalizedNodeId = nodeId.replace(/\\/g, '/')
```

**经验总结**：所有路径相关操作（缓存键、节点 ID、IPC 调用）必须使用统一的路径格式

---

### 坑4：IPC 序列化错误 "An object could not be cloned"

**问题**：保存配置时抛出 `An object could not be cloned` 错误

**现象**：配置切换、创建配置时失败，后端没有日志

**原因**：树节点包含循环引用或不可序列化的对象（如 Vue 响应式代理、函数等）

**解决方案**：
```typescript
// 深拷贝数据，移除不可序列化的字段
const serializableData = JSON.parse(JSON.stringify({
  id: data.id,
  name: data.name,
  version: data.version,
  createdAt: data.createdAt,
  updatedAt: data.updatedAt,
  tree: this.serializeTree(data.tree)
}))
```

**经验总结**：通过 IPC 传递的数据必须是可序列化的，使用 `JSON.parse(JSON.stringify())` 深拷贝可以移除不可序列化的字段

---

### 坑5：响应式更新失效

**问题**：添加源后，树视图不自动更新

**现象**：需要手动刷新页面才能看到新添加的源

**原因**：Store 状态更新后，组件没有响应式更新

**解决方案**：
1. 确保使用 Pinia Store 的响应式状态
2. 更新状态后，触发 Vue 的响应式更新
3. 使用 `store.$patch()` 或直接修改状态属性

**经验总结**：使用 Pinia Store 时，确保状态更新能够触发组件重新渲染

### 坑6：配置 ID 生成策略不当

**问题**：用户要求配置 ID 使用配置名称 + 4位哈希

**原实现**：`config-${Date.now()}`（时间戳）

**新实现**：
```typescript
const generateHash4 = (str: string): string => {
  let hash = 0
  for (let i = 0; i < str.length; i++) {
    const char = str.charCodeAt(i)
    hash = ((hash << 5) - hash) + char
    hash = hash & hash // 转换为32位整数
  }
  const hash4 = Math.abs(hash) % 10000
  return String(hash4).padStart(4, '0')
}

const id = `${name}-${generateHash4(name)}`
```

**经验总结**：配置 ID 应该基于业务需求设计，既要保证唯一性，又要便于识别
