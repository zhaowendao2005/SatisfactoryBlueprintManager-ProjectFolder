```typescript
// 提取目录路径（不含文件名）
function extractDirectoryPath(fullPath: string, levels: number): string {
  // 1. 统一路径分隔符
  const normalized = fullPath.replace(/\\/g, '/')
  
  // 2. 去掉文件名（最后一段）
  const parts = normalized.split('/').filter(p => p.length > 0)
  const dirParts = parts.slice(0, -1) // 去掉最后一个（文件名）
  
  // 3. 取最后 n 级目录
  if (dirParts.length <= levels) {
    return dirParts.join('/')
  }
  
  return dirParts.slice(-levels).join('/')
}
```

### 4. **筛选逻辑改造**

**当前**：单一的 `filterBlueprintsByTags(blueprints, activeTags, logicMode)`

**改为**：**保持单一筛选函数，但路径标签和用户标签合并处理**

```typescript
function filterBlueprintsByTags(
  blueprints: ActiveBlueprintNodeWithTags[],
  activeTags: Set<string>,        // 包含路径标签和用户标签
  logicMode: 'and' | 'or'         // 统一的逻辑模式
): ActiveBlueprintNodeWithTags[] {
  if (activeTags.size === 0) {
    return blueprints
  }

  return blueprints.filter((blueprint) => {
    // 获取蓝图的所有标签：路径标签 + 用户标签
    const pathTagId = blueprint.directoryPath 
      ? `path:${blueprint.directoryPath}` 
      : null
    const userTags = blueprint.tags || []
    const allBlueprintTags = pathTagId 
      ? [pathTagId, ...userTags] 
      : userTags

    const tagArray = Array.from(activeTags)

    if (logicMode === 'and') {
      // 与逻辑：蓝图必须包含所有选中的标签（路径+用户）
      return tagArray.every((tagId) => allBlueprintTags.includes(tagId))
    } else {
      // 或逻辑：蓝图只需包含至少一个选中的标签
      return tagArray.some((tagId) => allBlueprintTags.includes(tagId))
    }
  })
}
```

**关键点**：
- 路径标签和用户标签在筛选时合并为一个完整标签集合
- 使用统一的与/或逻辑，不分别计算

## ✅ 设计决策

### 决策 1：路径追踪级数
- **默认值**：3 级
- **配置方式**：在 ConfigManager 中添加 UI 配置（滑块，范围 1-6）
- **存储位置**：`ConfigFileData.pathTagLevels` 字段
- **示例**：级数 3 时，路径 `C:\A\B\C\D\file.sbp` → 目录标签 `B/C/D`

- **架构**：路径标签和用户标签使用**统一的与/或逻辑**

- **优势**：逻辑简单，用户容易理解

### 决策 3：路径标签筛选器交互
- **多选支持**：是，可以选择多个路径标签
- **筛选行为**：与用户标签一起，根据"与/或"逻辑计算
- **示例**：
  - 选择 `path:Factory` + `建筑` + 逻辑"与" → 筛选 Factory 目录下的建筑类蓝图
  - 选择 `path:Factory` + `path:Base` + 逻辑"或" → 筛选 Factory 或 Base 目录下的所有蓝图

### 工具函数设计

```typescript
// 1. 提取目录路径（不含文件名）
function extractDirectoryPath(fullPath: string, levels: number): string {
  const normalized = fullPath.replace(/\\/g, '/')
  const parts = normalized.split('/').filter(p => p.length > 0)
  const dirParts = parts.slice(0, -1) // 去掉文件名
  
  if (dirParts.length === 0) return ''
  if (dirParts.length <= levels) return dirParts.join('/')
  
  // 取最后 n 级目录
  const lastNLevels = dirParts.slice(-levels)
  
  // 溢出处理：如果超过 50 字符，前2段 + ... + 最后1段
  const joined = lastNLevels.join('/')
  if (joined.length <= 50) return joined
  
  if (lastNLevels.length >= 3) {
    return `${lastNLevels[0]}/${lastNLevels[1]}/.../${lastNLevels[lastNLevels.length - 1]}`
  }
  
  return joined
}

// 2. 提取所有唯一的目录路径标签
function extractPathTags(
  blueprints: ActiveBlueprintNodeWithTags[]
): TagDefinition[] {
  const pathSet = new Set<string>()
  
  for (const bp of blueprints) {
    if (bp.directoryPath) {
      pathSet.add(bp.directoryPath)
    }
  }
  
  return Array.from(pathSet).map((dirPath) => ({
    id: `path:${dirPath}`,
    name: dirPath,
    color: '#3B82F6', // 蓝色
  }))
}

// 3. 统一筛选函数
function filterBlueprintsByTags(
  blueprints: ActiveBlueprintNodeWithTags[],
  activeTags: Set<string>,
  logicMode: 'and' | 'or'
): ActiveBlueprintNodeWithTags[] {
  if (activeTags.size === 0) return blueprints

  return blueprints.filter((blueprint) => {
    // 合并路径标签和用户标签
    const pathTagId = blueprint.directoryPath ? `path:${blueprint.directoryPath}` : null
    const userTags = blueprint.tags || []
    const allTags = pathTagId ? [pathTagId, ...userTags] : userTags

    const tagArray = Array.from(activeTags)

    if (logicMode === 'and') {
      return tagArray.every((tagId) => allTags.includes(tagId))
    } else {
      return tagArray.some((tagId) => allTags.includes(tagId))
    }
  })
}
```

// 2. 修改 pathTagLevels
async setPathTagLevels(levels: number): Promise<void> {
  if (levels < 1 || levels > 6) {
    throw new Error('路径追踪级数必须在 1-6 之间')
  }
  
  this.pathTagLevels = levels
  
  // 重新计算所有蓝图的 directoryPath
  this.recalculateDirectoryPaths()
  
  await this.saveCurrentConfig()
}

// 3. 重新计算所有目录路径
recalculateDirectoryPaths(): void {
  const updatePaths = (nodes: ActiveBlueprintNode[]): void => {
    for (const node of nodes) {
      if (node.type === 'blueprint' && node.path) {
        node.directoryPath = extractDirectoryPath(node.path, this.pathTagLevels)
      }
      if (node.children) {
        updatePaths(node.children)
      }
    }
  }
  
  updatePaths(this.treeData)
}
