## **完整技术方案**

### **一、依赖选择与对比**

#### **方案对比表**

| 依赖库 | 优点 | 缺点 | 推荐度 |
|--------|------|------|--------|
| **@nut-tree/nut-js** | ✅ TypeScript 原生支持<br>✅ 现代 Promise API<br>✅ 跨平台<br>✅ 无需原生编译 | ❌ 社区较小<br>❌ 部分功能需额外配置 | ⭐⭐⭐⭐⭐ |
| **robotjs** | ✅ 社区成熟<br>✅ 功能全面<br>✅ 文档丰富 | ❌ 需要原生编译<br>❌ Electron 打包复杂<br>❌ 同步 API | ⭐⭐⭐ |
| **node-win32-api** | ✅ Windows API 直接调用<br>✅ 性能最好 | ❌ 仅 Windows<br>❌ API 复杂<br>❌ TypeScript 支持一般 | ⭐⭐ |

#### **最终推荐：@nut-tree/nut-js**

**理由：**
1. 你的项目已经用 TypeScript，nut-js 原生支持最好
2. 无需原生编译，Electron 打包更简单
3. Promise-based API 更适合现代异步操作
4. 如果未来需要跨平台，无需改代码

// public/types/automation/index.ts
/**
 * 自动化任务配置
 */
export interface AutomationTaskConfig {
  /** 任务 ID */
  id: string;
  /** 任务名称 */
  name: string;
  /** 任务描述 */
  description?: string;
  /** 是否启用 */
  enabled: boolean;
  /** 创建时间 */
  createdAt: number;
  /** 最后修改时间 */
  updatedAt: number;
  /** 执行步骤 */
  steps: AutomationStep[];
  /** 全局延迟配置 */
  globalDelays: DelayConfig;
}

/**
 * 执行步骤类型
 */
export type AutomationStep =
  | ClickStep
  | TypeTextStep
  | KeyPressStep
  | WaitStep
  | MoveMouseStep;

/**
 * 步骤基类
 */
interface BaseStep {
  /** 步骤 ID */
  id: string;
  /** 步骤名称 */
  name: string;
  /** 是否启用 */
  enabled: boolean;
  /** 步骤执行前延迟（毫秒） */
  delayBefore?: number;
  /** 步骤执行后延迟（毫秒） */
  delayAfter?: number;
}

/**
 * 点击步骤
 */
export interface ClickStep extends BaseStep {
  type: 'click';
  /** 点击坐标 X */
  x: number;
  /** 点击坐标 Y */
  y: number;
  /** 鼠标按钮：left | right | middle */
  button: 'left' | 'right' | 'middle';
  /** 点击次数（1=单击, 2=双击） */
  clickCount: 1 | 2;
}

/**
 * 输入文本步骤
 */
export interface TypeTextStep extends BaseStep {
  type: 'typeText';
  /** 要输入的文本 */
  text: string;
  /** 输入速度（字符间隔毫秒数） */
  typingSpeed: number;
}

/**
 * 按键步骤
 */
export interface KeyPressStep extends BaseStep {
  type: 'keyPress';
  /** 按键名称，如 'Enter', 'Escape', 'A' 等 */
  key: string;
  /** 是否按住 Ctrl */
  ctrl?: boolean;
  /** 是否按住 Alt */
  alt?: boolean;
  /** 是否按住 Shift */
  shift?: boolean;
}

/**
 * 等待步骤
 */
export interface WaitStep extends BaseStep {
  type: 'wait';
  /** 等待时长（毫秒） */
  duration: number;
}

/**
 * 移动鼠标步骤
 */
export interface MoveMouseStep extends BaseStep {
  type: 'moveMouse';
  /** 目标坐标 X */
  x: number;
  /** 目标坐标 Y */
  y: number;
  /** 移动速度（毫秒，越小越快） */
  speed: number;
}

/**
 * 全局延迟配置
 */
export interface DelayConfig {
  /** 步骤间默认延迟（毫秒） */
  betweenSteps: number;
  /** 点击前延迟（毫秒） */
  beforeClick: number;
  /** 点击后延迟（毫秒） */
  afterClick: number;
  /** 输入前延迟（毫秒） */
  beforeType: number;
  /** 输入后延迟（毫秒） */
  afterType: number;
  /** 按键前延迟（毫秒） */
  beforeKeyPress: number;
  /** 按键后延迟（毫秒） */
  afterKeyPress: number;
  /** 鼠标移动速度（毫秒） */
  mouseMoveDuration: number;
}

/**
 * 任务执行状态
 */
export interface TaskExecutionStatus {
  taskId: string;
  status: 'idle' | 'running' | 'paused' | 'completed' | 'error';
  currentStepIndex: number;
  totalSteps: number;
  startTime?: number;
  endTime?: number;
  error?: string;
}

/**
 * 坐标标定结果
 */
export interface CoordinateCalibration {
  x: number;
  y: number;
  timestamp: number;
  screenInfo: {
    width: number;
    height: number;
    scaleFactor: number;
  };
}

<script setup lang="ts">
import { ref, reactive } from 'vue';
import type { 
  AutomationTaskConfig, 
  AutomationStep,
  DelayConfig 
} from '@/types/automation';

// 默认配置
const defaultDelays: DelayConfig = {
  betweenSteps: 200,
  beforeClick: 100,
  afterClick: 200,
  beforeType: 100,
  afterType: 200,
  beforeKeyPress: 50,
  afterKeyPress: 100,
  mouseMoveDuration: 300
};

const taskConfig = reactive<AutomationTaskConfig>({
  id: generateId(),
  name: '',
  description: '',
  enabled: true,
  createdAt: Date.now(),
  updatedAt: Date.now(),
  steps: [],
  globalDelays: { ...defaultDelays }
});

// 步骤编辑相关
const stepDialogVisible = ref(false);
const editingStep = ref<AutomationStep | null>(null);
const editingStepIndex = ref<number>(-1);

function showAddStepDialog() {
  editingStep.value = null;
  editingStepIndex.value = -1;
  stepDialogVisible.value = true;
}

function editStep(index: number) {
  editingStep.value = { ...taskConfig.steps[index] };
  editingStepIndex.value = index;
  stepDialogVisible.value = true;
}

function removeStep(index: number) {
  taskConfig.steps.splice(index, 1);
}

function handleStepConfirm(step: AutomationStep) {
  if (editingStepIndex.value >= 0) {
    taskConfig.steps[editingStepIndex.value] = step;
  } else {
    taskConfig.steps.push(step);
  }
}

// 工具函数
function getStepTypeName(type: string): string {
  const map: Record<string, string> = {
    click: '点击',
    typeText: '输入文本',
    keyPress: '按键',
    wait: '等待',
    moveMouse: '移动鼠标'
  };
  return map[type] || type;
}

function getStepTypeColor(type: string): string {
  const map: Record<string, string> = {
    click: 'primary',
    typeText: 'success',
    keyPress: 'warning',
    wait: 'info',
    moveMouse: 'default'
  };
  return map[type] || 'default';
}

function generateId(): string {
  return `task_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
}

async function saveConfig() {
  // 调用 IPC 保存配置
  await window.electron.automation.saveTask(taskConfig);
}

async function testRun() {
  // 调用 IPC 测试运行
  await window.electron.automation.executeTask(taskConfig.id);
}
</script>

// src-electron/Service/Automation/automation-service.ts
import { mouse, keyboard, Key, Button, straightTo, sleep } from '@nut-tree/nut-js';
import { screen, BrowserWindow } from 'electron';
import type {
  AutomationTaskConfig,
  AutomationStep,
  ClickStep,
  TypeTextStep,
  KeyPressStep,
  WaitStep,
  MoveMouseStep,
  TaskExecutionStatus,
  DelayConfig
} from '../../../public/types/automation';

export class AutomationService {
  private currentTask: TaskExecutionStatus | null = null;
  private shouldStop = false;
  private overlayWindow: BrowserWindow | null = null;

  /**
   * 设置悬浮窗引用（用于执行时隐藏）
   */
  setOverlayWindow(window: BrowserWindow) {
    this.overlayWindow = window;
  }

  /**
   * 执行自动化任务
   */
  async executeTask(config: AutomationTaskConfig): Promise<void> {
    if (this.currentTask?.status === 'running') {
      throw new Error('已有任务正在执行中');
    }

    this.shouldStop = false;
    this.currentTask = {
      taskId: config.id,
      status: 'running',
      currentStepIndex: 0,
      totalSteps: config.steps.filter(s => s.enabled).length,
      startTime: Date.now()
    };

    try {
      // 隐藏悬浮窗
      if (this.overlayWindow && !this.overlayWindow.isDestroyed()) {
        this.overlayWindow.hide();
      }

      // 初始延迟
      await sleep(500);

      // 执行每个步骤
      for (let i = 0; i < config.steps.length; i++) {
        if (this.shouldStop) {
          this.currentTask.status = 'paused';
          break;
        }

        const step = config.steps[i];
        if (!step.enabled) continue;

        this.currentTask.currentStepIndex = i;

        // 执行前延迟
        const delayBefore = step.delayBefore ?? this.getStepDefaultDelay(step.type, config.globalDelays, 'before');
        if (delayBefore > 0) {
          await sleep(delayBefore);
        }

        // 执行步骤
        await this.executeStep(step, config.globalDelays);

        // 执行后延迟
        const delayAfter = step.delayAfter ?? this.getStepDefaultDelay(step.type, config.globalDelays, 'after');
        if (delayAfter > 0) {
          await sleep(delayAfter);
        }

        // 步骤间延迟
        if (i < config.steps.length - 1 && config.globalDelays.betweenSteps > 0) {
          await sleep(config.globalDelays.betweenSteps);
        }
      }

      this.currentTask.status = 'completed';
      this.currentTask.endTime = Date.now();
    } catch (error) {
      this.currentTask.status = 'error';
      this.currentTask.error = (error as Error).message;
      throw error;
    } finally {
      // 恢复悬浮窗
      if (this.overlayWindow && !this.overlayWindow.isDestroyed()) {
        await sleep(300);
        this.overlayWindow.show();
      }
    }
  }

  /**
   * 执行单个步骤
   */
  private async executeStep(step: AutomationStep, globalDelays: DelayConfig): Promise<void> {
    switch (step.type) {
      case 'click':
        await this.executeClickStep(step as ClickStep);
        break;
      case 'typeText':
        await this.executeTypeTextStep(step as TypeTextStep);
        break;
      case 'keyPress':
        await this.executeKeyPressStep(step as KeyPressStep);
        break;
      case 'wait':
        await this.executeWaitStep(step as WaitStep);
        break;
      case 'moveMouse':
        await this.executeMoveMouseStep(step as MoveMouseStep, globalDelays);
        break;
      default:
        console.warn(`未知的步骤类型: ${(step as any).type}`);
    }
  }

  /**
   * 执行点击步骤
   */
  private async executeClickStep(step: ClickStep): Promise<void> {
    await mouse.setPosition({ x: step.x, y: step.y });
    await sleep(50);

    const button = this.mapButton(step.button);
    
    if (step.clickCount === 2) {
      await mouse.doubleClick(button);
    } else {
      await mouse.click(button);
    }
  }

  /**
   * 执行输入文本步骤
   */
  private async executeTypeTextStep(step: TypeTextStep): Promise<void> {
    if (step.typingSpeed > 0) {
      // 逐字符输入
      for (const char of step.text) {
        await keyboard.type(char);
        await sleep(step.typingSpeed);
      }
    } else {
      // 瞬间输入
      await keyboard.type(step.text);
    }
  }

  /**
   * 执行按键步骤
   */
  private async executeKeyPressStep(step: KeyPressStep): Promise<void> {
    const keys: Key[] = [];

    // 添加组合键
    if (step.ctrl) keys.push(Key.LeftControl);
    if (step.alt) keys.push(Key.LeftAlt);
    if (step.shift) keys.push(Key.LeftShift);

    // 添加主键
    const mainKey = this.mapKey(step.key);
    keys.push(mainKey);

    // 按下所有键
    for (const key of keys) {
      await keyboard.pressKey(key);
    }

    await sleep(50);

    // 释放所有键（逆序）
    for (let i = keys.length - 1; i >= 0; i--) {
      await keyboard.releaseKey(keys[i]);
    }
  }

  /**
   * 执行等待步骤
   */
  private async executeWaitStep(step: WaitStep): Promise<void> {
    await sleep(step.duration);
  }

  /**
   * 执行移动鼠标步骤
   */
  private async executeMoveMouseStep(step: MoveMouseStep, globalDelays: DelayConfig): Promise<void> {
    const speed = step.speed || globalDelays.mouseMoveDuration;
    
    if (speed > 0) {
      // 平滑移动
      mouse.config.mouseSpeed = speed;
      await mouse.move(straightTo({ x: step.x, y: step.y }));
    } else {
      // 瞬间移动
      await mouse.setPosition({ x: step.x, y: step.y });
    }
  }

  /**
   * 停止任务执行
   */
  stopTask(): void {
    this.shouldStop = true;
  }

  /**
   * 获取当前任务状态
   */
  getTaskStatus(): TaskExecutionStatus | null {
    return this.currentTask;
  }

  /**
   * 获取步骤默认延迟
   */
  private getStepDefaultDelay(
    stepType: string,
    globalDelays: DelayConfig,
    timing: 'before' | 'after'
  ): number {
    const map: Record<string, { before: number; after: number }> = {
      click: { before: globalDelays.beforeClick, after: globalDelays.afterClick },
      typeText: { before: globalDelays.beforeType, after: globalDelays.afterType },
      keyPress: { before: globalDelays.beforeKeyPress, after: globalDelays.afterKeyPress },
      wait: { before: 0, after: 0 },
      moveMouse: { before: 0, after: 0 }
    };

    return map[stepType]?.[timing] ?? 0;
  }

  /**
   * 映射鼠标按钮
   */
  private mapButton(button: string): Button {
    switch (button) {
      case 'left': return Button.LEFT;
      case 'right': return Button.RIGHT;
      case 'middle': return Button.MIDDLE;
      default: return Button.LEFT;
    }
  }

  /**
   * 映射键盘按键
   */
  private mapKey(keyName: string): Key {
    const upperKey = keyName.toUpperCase();
    
    // 常用按键映射
    const keyMap: Record<string, Key> = {
      'ENTER': Key.Return,
      'RETURN': Key.Return,
      'ESC': Key.Escape,
      'ESCAPE': Key.Escape,
      'TAB': Key.Tab,
      'SPACE': Key.Space,
      'BACKSPACE': Key.Backspace,
      'DELETE': Key.Delete,
      'UP': Key.Up,
      'DOWN': Key.Down,
      'LEFT': Key.Left,
      'RIGHT': Key.Right,
      'HOME': Key.Home,
      'END': Key.End,
      'PAGEUP': Key.PageUp,
      'PAGEDOWN': Key.PageDown,
      // 功能键
      'F1': Key.F1, 'F2': Key.F2, 'F3': Key.F3, 'F4': Key.F4,
      'F5': Key.F5, 'F6': Key.F6, 'F7': Key.F7, 'F8': Key.F8,
      'F9': Key.F9, 'F10': Key.F10, 'F11': Key.F11, 'F12': Key.F12,
    };

    if (keyMap[upperKey]) {
      return keyMap[upperKey];
    }

    // 字母和数字键
    if (upperKey.length === 1) {
      return Key[upperKey as keyof typeof Key] || Key.A;
    }

    return Key.A; // 默认值
  }
}

// 导出单例
export const automationService = new AutomationService();

      // 定期发送鼠标位置
      mouseTrackingInterval = setInterval(() => {

        const display = screen.getDisplayNearestPoint(point);

        const calibration: CoordinateCalibration = {
          x: point.x,
          y: point.y,
          timestamp: Date.now(),
          screenInfo: {
            width: display.bounds.width,
            height: display.bounds.height,
            scaleFactor: display.scaleFactor
          }
        };

        // 清理
        clearInterval(mouseTrackingInterval);
        globalShortcut.unregister('F8');

        event.sender.send('automation:calibration:complete', calibration);
        resolve(calibration);
      });

      if (!ret) {
        clearInterval(mouseTrackingInterval);
        resolve({
          x: 0,
          y: 0,
          timestamp: Date.now(),
          screenInfo: { width: 0, height: 0, scaleFactor: 1 }
        });
      }
    });
  });

  // 停止标定
  ipcMain.handle('automation:stopCalibration', () => {
    globalShortcut.unregister('F8');
  });

  // 停止任务
  ipcMain.handle('automation:stopTask', () => {
    automationService.stopTask();
  });

  // 获取任务状态
  ipcMain.handle('automation:getTaskStatus', () => {
    return automationService.getTaskStatus();
  });
}
